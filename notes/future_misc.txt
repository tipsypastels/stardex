  import { parsePBSAsRecords } from "../src/models/pokemon/pbs/parse.ts";

  const records = readdirSync("samples/pbs")
    .filter((path) => /^pokemon.*.txt$/.test(path))

    .flatMap((file) => parsePBSAsRecords(file).out);

  const buckets = getPBSFormFilterBuckets(records);

  writeFileSync("samples/pbs/buckets.json", JSON.stringify(buckets, null, 2));
